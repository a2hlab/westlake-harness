 1214 // [S23 cut5 2026-07-09] Seed the geometry of the EXACT OHNativeWindow instance
 1215 // Unity dequeues from, on the last-mile handoff. cut4 proved SET_BUFFER_GEOMETRY
 1216 // (rc=0) reaches window->config, but the bridge poked a sibling nw
 1217 // (oh_rs_get_child_surface_window's 0x..B50) while Unity uses the instance stamped
 1218 // into Surface.mNativeObject / returned by ANativeWindow_fromSurface (0x..A20).
 1219 // native_window.cpp:229 NativeWindowRequestBuffer reads THIS instance's config;
 1220 // if it is 0x0 the server allocs Buffer[0 0] -> NO_BUFFER 50002000 storm, no pixels.
 1221 // Poke here (creation/stamp + handoff) so the dequeued instance carries 1200x1920
 1222 // BEFORE Unity's first RequestBuffer. 5bb5 panel = 1200x1920 (hard fallback; never
 1223 // 0/1). Log the instance address so "same instance as Unity RequestBuffer nw_B" is
 1224 // nailed in the log.
 1225 extern "C" void nwfs_seed_window_geometry(void* nw, const char* where) {
 1226     if (!nw) return;
 1227     int32_t w = 1200, h = 1920;  // TODO: prefer live display bounds; 5bb5 panel is 1200x1920
 1228     // [S23 cut5.1] nw handed to Unity is the AdapterAnw shim (AOSP ABI), NOT a raw
 1229     // OHNativeWindow. SET_BUFFER_GEOMETRY on the shim -> rc=0x40001000 (invalid arg).
 1230     // Unwrap to the shim's embedded real OHNativeWindow first (same pattern as
 1231     // ANativeWindow_getWidth line ~1140). This is the SAME underlying producer
 1232     // Unity's RequestBuffer (shim wrapper -> OH_NativeWindow_NativeWindowRequestBuffer)
 1233     // dequeues from, so config now reaches the storming path.
 1234     OHNativeWindow* oh = anw_unwrap(reinterpret_cast<ANativeWindow*>(nw));
 1235     int32_t rc = NativeWindowHandleOpt(oh, OH_OP_SET_BUFFER_GEOMETRY, w, h);
 1236     HiLogPrint(3, 4, 0xD000F00u, "OH_NWSeed",
 1237                "[%{public}s] SET_BUFFER_GEOMETRY shim=%{public}p oh=%{public}p %{public}dx%{public}d rc=%{public}d",
 1238                where, nw, (void*)oh, w, h, rc);

 1241 extern "C" ANativeWindow* ANativeWindow_fromSurface(JNIEnv* env, jobject surface) {
 1242     NWFS_INFO("[STAGE0] ENTER env=%{public}p surface=%{public}p",
 1243               (void*)env, (void*)surface);
 1244     if (!env || !surface) {
 1245         NWFS_WARN("[STAGE0] null arg -> null");
 1246         return nullptr;
 1247     }
 1248     jclass cls = env->FindClass("android/view/Surface");
 1249     if (!cls) { env->ExceptionClear();
 1250         NWFS_WARN("[STAGE0] FindClass failed -> null"); return nullptr; }
 1251     jfieldID fid = env->GetFieldID(cls, "mNativeObject", "J");
 1252     env->DeleteLocalRef(cls);
 1253     if (!fid) { env->ExceptionClear();
 1254         NWFS_WARN("[STAGE0] GetFieldID failed -> null"); return nullptr; }
 1255     jlong nativeObj = env->GetLongField(surface, fid);
 1256     NWFS_INFO("[STAGE0] mNativeObject=0x%{public}llx",
 1257               (unsigned long long)nativeObj);
 1258     if (nativeObj == 0) {
 1259         NWFS_WARN("[STAGE0] mNativeObject==0 -> null");
 1260         return nullptr;
 1261     }
 1262     if (is_sentinel_handle(nativeObj)) {
 1263         NWFS_WARN("[STAGE0] sentinel handle 0x%{public}llx -> null "
 1264                   "(no real BBQ-bound surface yet; hwui will retry)",
 1265                   (unsigned long long)nativeObj);
 1266         return nullptr;
 1267     }
 1268     // Real OHNativeWindow* set by BBQ_nativeGetSurface above. On OH,
 1269     // OHNativeWindow ≡ ANativeWindow, so hand it back to hwui directly.
 1270     // [S23 cut5 PRIMARY] last-mile: this is the exact instance Unity dequeues
 1271     // from — seed its geometry before returning so its first RequestBuffer
 1272     // carries 1200x1920 (not Buffer[0 0]). Instance-identity nailed via the log.
 1273     nwfs_seed_window_geometry(reinterpret_cast<void*>(nativeObj), "STAGE0-handoff");
 1274     NWFS_INFO("[STAGE0] returning OHNativeWindow=%{public}p directly",
 1275               reinterpret_cast<void*>(nativeObj));
 1276     return reinterpret_cast<ANativeWindow*>(nativeObj);
 1277 }
