768: // Helper: register one method-array, method-by-method, tolerate failures
769: // =====================================================================
770: int registerOne(JNIEnv* env, const char* className,
771:                 const JNINativeMethod* methods, size_t count) {
772:     jclass clazz = env->FindClass(className);
773:     if (!clazz) {
774:         ALOGW("FindClass(%s) failed — class not in BCP yet?", className);
775:         if (env->ExceptionCheck()) env->ExceptionClear();
776:         return -1;
777:     }
778:     int ok = 0, fail = 0;
779:     for (size_t i = 0; i < count; ++i) {
780:         jint rc = env->RegisterNatives(clazz, &methods[i], 1);
781:         if (rc == JNI_OK) {
782:             ++ok;
783:         } else {
784:             ++fail;
785:             ALOGW("compat_shim: register %s.%s%s failed",
786:                   className, methods[i].name, methods[i].signature);
787:             if (env->ExceptionCheck()) env->ExceptionClear();
788:         }
789:     }
790:     env->DeleteLocalRef(clazz);
791:     ALOGI("compat_shim: %s registered %d/%zu", className, ok, count);
792:     return 0;  // never propagate failure to caller — shim is best-effort
793: }
