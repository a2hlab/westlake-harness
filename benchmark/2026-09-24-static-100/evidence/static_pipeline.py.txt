"""Static half of the 100-app loop: scan -> oh-resolve -> gap-map for every app in a corpus file.

usage: static_pipeline.py <corpus.json> <static dir> [--jobs N]
corpus.json: {"apps": {key: {"input": path to .apk/.xapk, "stack": ..., "source": ...}}}
The provider is this workstation's westlake main source build; native imports resolve against the OH 6.1.0.31
SDK's public libraries, the link-only stubs of the 47 pinned firmware libraries, and the built runtime.
Every step is skipped when its output exists, so the script is resumable.
"""
import argparse
import json
import subprocess
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

A = Path("/home/dspfac/a2hlab/source-closure/verify"); O = A / "out"
HARNESS = Path("/Users/zhaoyue/orca/workspaces/westlake-harness")
TOOL = str(Path.home() / "a2hlab/harness-venv/bin/westlake-apk-gap")
SDK_LIBS = A / "toolchains/ohos-sdk/native/sysroot/usr/lib/aarch64-linux-ohos"
LIB_DIRS = [SDK_LIBS, O / "native-imports", O / "native-runtime"]
NDK_API = A / "toolchains/ohos-sdk/native/sysroot/usr/lib/aarch64-linux-ohos"


def run(cmd, log):
    with open(log, "w") as out:
        return subprocess.run(cmd, stdout=out, stderr=subprocess.STDOUT, cwd=HARNESS).returncode


def one(key, app, d):
    logs = d / "logs"; scan = d / "scans" / f"{key}.json"; oh = d / "oh" / f"{key}.json"; gm = d / "maps" / key
    if not scan.exists() and run([TOOL, "scan", app["input"], "--runtime", str(d / "runtime-index.json"),
                                  "--target-abi", "arm64-v8a", "--out", str(scan)], logs / f"{key}.scan.log"):
        return key, "scan failed"
    abi = json.loads(scan.read_text())["inventory"]["native_resolution"]["abi_status"]
    if abi == "target-abi-unavailable":
        return key, "32-bit only: excluded"
    if not oh.exists():
        cmd = [TOOL, "oh-resolve", "--scan", str(scan), "--app-key", key, "--board", "OH 6.1.0.31 SDK + firmware stubs + westlake main runtime", "--out", str(oh)]
        for lib in LIB_DIRS: cmd += ["--lib-dir", str(lib)]
        if run(cmd, logs / f"{key}.oh.log"):
            return key, "oh-resolve failed"
    if not (gm / "gap-map.json").exists():
        cmd = [TOOL, "gap-map", "--scan", str(scan), "--apk", app["input"], "--aosp", str(A / "android-source"),
               "--westlake", str(A / "westlake"), "--westlake-label", "westlake main 22b9453 (source build)",
               "--manifest-repo", str(Path.home() / "a2hlab/manifest"), "--oh-resolution", str(oh), "--app-key", key,
               "--board-libs", str(HARNESS / "benchmark/2026-09-18-oh-board/oh-board-libraries.txt"),
               "--runtime-libs", str(O / "native-runtime"), "--out", str(gm)]
        if run(cmd, logs / f"{key}.gapmap.log"):
            return key, "gap-map failed"
    rows = json.loads((gm / "gap-map.json").read_text())
    return key, f"ok ({abi})"


def main():
    p = argparse.ArgumentParser(); p.add_argument("corpus"); p.add_argument("static_dir"); p.add_argument("--jobs", type=int, default=4)
    a = p.parse_args(); d = Path(a.static_dir)
    for sub in ("scans", "oh", "maps", "logs"): (d / sub).mkdir(parents=True, exist_ok=True)
    apps = json.loads(Path(a.corpus).read_text())["apps"]
    with ThreadPoolExecutor(a.jobs) as pool:
        for key, status in pool.map(lambda kv: one(kv[0], kv[1], d), apps.items()):
            print(f"{key:28s} {status}", flush=True)


if __name__ == "__main__":
    main()
