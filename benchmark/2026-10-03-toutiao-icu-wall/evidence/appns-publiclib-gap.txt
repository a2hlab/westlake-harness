## app ns construction (bms/src/adapter/framework/app-native-loader/src/app_native_loader.c)
create_configured_namespaces(..., bridge_search_paths, bridge_permitted_paths, bridge_shared_sonames,
                             bridge_bootstrap_soname, app_search_paths, app_permitted_paths)
 - app_search_paths/app_permitted_paths = the apk's own libs (/data/app/.../<pkg>/lib/arm64-v8a).
 - bridge_shared_sonames = soname allowlist, passed ONLY for the bridge ns, not a full app public-lib list.
 There is no argument carrying the AOSP public-library set into the app ns, so the app ns can name only (a) its own
 apk libs and (b) whatever is transitively pulled via the libs it does link. Any NDK public lib the app dlopen()s by
 soname that is not transitively present -> ENOENT.

## WHY libicuuc.so resolves but libicui18n.so ENOENTs (same ns, same /system/android/lib64 dir) -- DT_NEEDED pull:
# (llvm-objdump -p, U13 payload native/payload/android/lib64)
#   liboh_android_runtime.so   NEEDED(icu): libicuuc.so 
#   libandroidfw.so            NEEDED(icu): libicuuc.so 
#   libminikin.so              NEEDED(icu): libicuuc.so libicui18n.so 
#   libicu_jni.so              NEEDED(icu): libicuuc.so libicui18n.so 
#   libicui18n.so              NEEDED(icu): libicuuc.so 
# libicuuc.so is DT_NEEDED by liboh_android_runtime + libandroidfw + libminikin (all reachable from the app), so it is
# pulled into the app-reachable graph and dlopen("libicuuc.so") hits the already-loaded copy. libicui18n.so is NEEDED
# only by libminikin + libicu_jni, which load in the BRIDGE/runtime ns; their copy of libicui18n is not re-exported to
# the app ns by soname, and the app ns has no public-lib entry for it -> dlopen("libicui18n.so") = ENOENT. libicuuc
# masking the gap is luck of the dependency graph, not a deliberate ICU exposure.

## The runtime DOES load both ICU libs -- but RTLD_LOCAL, in its own ns (android-runtime/src/westlake_icu_overrides.cpp)
#   dlopen("/system/android/lib64/libicuuc.so",  RTLD_NOW|RTLD_LOCAL)   (line 378)
#   dlopen("/system/android/lib64/libicui18n.so", RTLD_NOW|RTLD_LOCAL)  (line 385)
# RTLD_LOCAL => not added to the global scope; binds the _72 cohort for ART. These copies exist in the process but are
# not nameable from the app ns.

## WALL 2 (behind the ns wall): symbol VERSIONING. toutiao dlsym()s UNVERSIONED names; our ICU exports only _72.
#   libicuuc.so exports for 'ubrk_open':
#     ubrk_open_72
#   libicuuc.so exports for 'u_charType':
#     u_charType_72
#   libicuuc.so exports for 'uloc_setDefault':
#     uloc_setDefault_72
# => only ubrk_open_72 / u_charType_72 / uloc_setDefault_72 exist; dlsym(handle,"ubrk_open") returns NULL.
# So even if libicui18n.so is made visible to the app ns, toutiao's unversioned dlsym still returns null -> the SAME
# ICUWrapper ctor 'blr x8' with x8=0 -> pc=0. The deployed ICU (U10.1T8 and U13 alike) is built WITH symbol renaming
# (U_DISABLE_RENAMING=0); toutiao expects unrenamed ICU (or its own libkryptoni18n.so, which this apk does not bundle).
